"""fs.stale: which folders inside a directory have been left alone, and for how long."""
import os
import re
import time

from los import CommandError

PERIODS = {"minute": 60, "hour": 3600, "day": 86400, "week": 7 * 86400, "month": 30 * 86400, "year": 365 * 86400}
DEFAULT = "6 months"   # untouched this long counts as abandoned
SHOWN = 50             # folders listed before the rest is counted instead


def _age(text):
    """Seconds, from text such as 30 days, 6 months or 1 year."""
    match = re.fullmatch(r"\s*(\d+(?:\.\d+)?)\s*([a-z]+?)s?\s*", text.lower())
    periods = [period for period in PERIODS if match and period.startswith(match[2])]
    if len(periods) != 1:
        raise CommandError(f"an age looks like 6 months or 1 year, not {text!r}")
    return float(match[1]) * PERIODS[periods[0]]


def _last_change(folder):
    """When anything in the folder last changed: its newest file, or the folder itself if it has none."""
    newest = 0
    for parent, _, files in os.walk(folder):
        for file in files:
            try:
                newest = max(newest, os.lstat(os.path.join(parent, file)).st_mtime)
            except OSError:
                pass
    return newest or os.lstat(folder).st_mtime


def _ago(seconds):
    days = int(seconds // 86400)
    if days < 60:
        return f"{days} day{'' if days == 1 else 's'}"
    if days < 730:
        return f"{days // 30} months"
    return f"{days // 365} years"


def do_stale(path=None, older_than=None):
    if not path or not os.path.isdir(path):
        raise CommandError(f"{path} is not a directory")
    limit = older_than or DEFAULT
    now = time.time()
    cutoff = now - _age(limit)
    # Each folder directly inside counts as one project; hidden ones such as .cache do not.
    projects = sorted((_last_change(entry.path), entry.name) for entry in os.scandir(path)
                      if entry.is_dir(follow_symlinks=False) and not entry.name.startswith("."))
    if not projects:
        return f"{path} has no project folders in it."
    stale = [(changed, name) for changed, name in projects if changed <= cutoff]
    if not stale:
        changed, name = projects[0]
        return (f"No folder in {path} has gone untouched for {limit} ({len(projects)} looked at). "
                f"The quietest is {name}, last changed {_ago(now - changed)} ago.")
    lines = [f"{len(stale)} of {len(projects)} folders in {path} untouched for {limit} or more, longest first:"]
    for changed, name in stale[:SHOWN]:
        day = time.strftime("%Y-%m-%d", time.localtime(changed))
        lines.append(f"{_ago(now - changed):>10}  {day}  {name}/")
    if len(stale) > SHOWN:
        lines.append(f"... and {len(stale) - SHOWN} more")
    return "\n".join(lines)

"""Project commands: how recently each project in a folder was worked on."""
import os
import time

from los import CommandError, judge

STATES = ["active", "dormant", "abandoned"]
QUESTION = "Nothing in a project has changed for this long. Is the project active, dormant or abandoned?"


def _newest(folder):
    """The latest modification time of the folder or of anything beneath it."""
    newest = os.lstat(folder).st_mtime
    for parent, folders, files in os.walk(folder):
        for name in folders + files:
            try:
                newest = max(newest, os.lstat(os.path.join(parent, name)).st_mtime)
            except OSError:
                pass
    return newest


def do_activity(path=None, only=None):
    if not path or not os.path.isdir(path):
        raise CommandError(f"{path} is not a directory")
    only = only.strip().lower() if only else None
    if only and only not in STATES:
        raise CommandError(f"--only is active, dormant or abandoned, not {only!r}")
    now = time.time()
    projects = []   # (whole days since the last change, name); hidden folders are not projects
    for entry in os.scandir(path):
        if entry.is_dir(follow_symlinks=False) and not entry.name.startswith("."):
            days = max(0, int((now - _newest(entry.path)) // 86400))
            projects.append((days, entry.name))
    if not projects:
        return f"{path} has no project folders in it."
    projects.sort()     # most recently changed first
    lines, counts = [], dict.fromkeys(STATES, 0)
    for days, name in projects:
        value = f"{days} days"
        # Where active ends and abandoned begins is the user's opinion, so it is judged, not coded.
        state = judge("activity", QUESTION, value, STATES)
        counts[state] += 1
        if not only or state == only:
            lines.append(f"{state:<10}{value:>10}  {name}")
    lines.append(", ".join(f"{counts[state]} {state}" for state in STATES))
    return "\n".join(lines)

import re

def rule(value):
    # Expects text like "90 days"; anything else goes back to the student.
    if not isinstance(value, str):
        return None
    m = re.fullmatch(r"\s*(\d+) days?\s*", value)
    if not m:
        return None
    days = int(m.group(1))
    if days <= 90:
        return "active"
    if days <= 364:
        return "dormant"
    return "abandoned"
